float3 ColorAdjust(float3 color)
{
    // ƒRƒ“ƒgƒ‰ƒXƒg
    const float middleGray = 0.18f;
    color = middleGray * pow(max(color, 0.0f) / middleGray, contrast);

    // ƒJƒ‰[ƒtƒBƒ‹ƒ^
    color *= colorFilter;

    // Ê“x
    float luma = dot(color, float3(0.2126f, 0.7152f, 0.0722f));
    color = lerp(luma.xxx, color, saturation);

    return max(color, 0.0f);
}

// ƒrƒlƒbƒg
float3 ApplyVignette(float3 color, float2 uv)
{
    // ’†S‚©‚ç‚Ì‹——£‚É‹­‚³‚ğŠ|‚¯‚é
    float2 d = abs(uv - 0.5f) * vignetteIntensity;

    // c‰¡”ä‚Å•â³‚µ‚ÄA‰¡’·‚Ì‰æ–Ê‚Å‚à‰~‚É‚·‚é
    d.x *= aspectRatio;

    // ’†S‚Ù‚Ç 1A‹÷‚Ù‚Ç 0 ‚É‚È‚é
    float factor = pow(saturate(1.0f - dot(d, d)), vignetteSmoothness);

    // ‹÷‚Ù‚Çƒrƒlƒbƒg‚ÌF‚É‹ß‚Ã‚¯‚é
    return color * lerp(vignetteColor, 1.0f, factor);
}

// ACES ƒg[ƒ“ƒ}ƒbƒv
float3 TonemapACES(float3 x)
{
    const float a = 2.51f;
    const float b = 0.03f;
    const float c = 2.43f;
    const float d = 0.59f;
    const float e = 0.14f;
    return saturate((x * (a * x + b)) / (x * (c * x + d) + e));
}

// ƒŠƒjƒA ‚©‚ç sRGB
float3 LinearToSRGB(float3 c)
{
    c = saturate(c);
    float3 low = c * 12.92f;
    float3 high = 1.055f * pow(c, 1.0f / 2.4f) - 0.055f;
    return (c <= 0.0031308f) ? low : high; // ¬•ª‚²‚Æ‚É‘I‚Ô
}

    float3 color = sceneTexture.Load(int3(pin.position.xy, 0)).rgb;
        
    // ˜Ioİ’è
    color *= exposure;

    // ƒJƒ‰[’²®

struct PS_OUT
{
    float4 albedo : SV_TARGET0;
    float4 normal : SV_TARGET1;
    float4 position : SV_TARGET2;
    float4 roughnessMetallic : SV_TARGET3;
    float4 emissive : SV_TARGET4;
    float occlusion : SV_TARGET5;

PS_OUT main(VS_OUT pin)
{
    PS_OUT pout;

    // AlbedoiƒKƒ“ƒ}•â³‚ÅƒŠƒjƒA‹óŠÔ‚Öj
    float4 baseColor = baseMap.Sample(samplerState, pin.texcoord);

    baseColor.a *= baseColorFactor.a;

    // MaskƒNƒŠƒbƒv

    baseColor.rgb *= baseColorFactor.rgb;
    pout.albedo = baseColor;
    
    // –@ü
    float3 N = normalize(pin.normal);
    pout.normal = float4(N * 0.5f + 0.5f, 1.0f);

    // ƒ[ƒ‹ƒhÀ•W
    pout.position = float4(pin.worldPos, 1.0f);

    return pout;
}

static const float2 RADIAL_BLUR_CENTER = float2(0.5f, 0.5f); // ’†S

float InterleavedGradientNoise(float2 pixel)
{
    return frac(52.9829189f * frac(dot(pixel, float2(0.06711056f, 0.00583715f))));
}

float4 main(VS_OUT pin) : SV_TARGET
{
    float2 uv = pin.texcoord;
    
    // ’†S‚©‚ç‚Ì‹——£
    float2 d = (uv - RADIAL_BLUR_CENTER) * 2.0f;
    d.x *= aspectRatio;
    float dist = length(d);
    
    // ŠŠ‚ç‚©‚É‚·‚é
    float mask = smoothstep(radialBlurRadius, radialBlurRadius + radialBlurSoftness, dist);

    // ü‚Ì‘S’·
    float2 blurVector = (RADIAL_BLUR_CENTER - uv) * radialBlurStrength * mask;
    
    // 1“_‚¸‚Âi‚Ş•
    float2 stepUV = blurVector / RADIAL_BLUR_SAMPLES;
    
    // “Ç‚İn‚ß‚ğ‹æŠÔ‚Ì’†‚Å‚¸‚ç‚·—Ê
    float jitter = InterleavedGradientNoise(pin.position.xy);

    float3 color = 0.0f;
    [unroll]
    for (int i = 0; i < RADIAL_BLUR_SAMPLES; i++)
    {
        color += sceneTexture.SampleLevel(linearSampler, uv + stepUV * (i + jitter), 0).rgb;
    }
    color /= RADIAL_BLUR_SAMPLES;

    return float4(color, 1.0f);
}

// irradianceMap: –‘Oô‚İ‚İÏ‚İ‚ÌDiffuseŠÂ‹«ƒ}ƒbƒv
// N            : –@üƒxƒNƒgƒ‹iƒ[ƒ‹ƒh‹óŠÔj
float3 IBL_Diffuse(
    TextureCube irradianceMap,
    SamplerState samplerState,
    float3 N,
    float3 diffuse_reflectance,
    float3 F0,
    float3 V)
{
    // ƒtƒŒƒlƒ‹€
    float NdotV = max(0.0001f, dot(N, V));
    float3 F = F0 + (1.0f - F0) * pow(clamp(1.0f - NdotV, 0.0f, 1.0f), 5.0f);
    
    // kd
    float3 kd = (1.0f - F);

    // –@ü•ûŒü‚ÅIrradianceƒ}ƒbƒv‚ğƒTƒ“ƒvƒŠƒ“ƒO
    float3 irradiance = irradianceMap.Sample(samplerState, N).rgb;

    return kd * diffuse_reflectance * irradiance;
}

// Specular IBL
// prefilteredMap: roughness‚²‚Æ‚ÉMip‚ª‚ ‚é–‘OƒtƒBƒ‹ƒ^ƒŠƒ“ƒOÏ‚İƒ}ƒbƒv
// brdfLut       : GGX BRDF‚ÌLookUpTable
// N             : –@üƒxƒNƒgƒ‹
// V             : ‹üƒxƒNƒgƒ‹
// F0            : ‚’¼“üË‚ÌƒtƒŒƒlƒ‹”½Ë—¦
// roughness     : ‘e‚³
float3 IBL_Specular(
    TextureCube prefilteredMap,
    Texture2D brdfLut,
    SamplerState samplerState,
    float3 N,
    float3 V,
    float3 F0,
    float roughness)
{
    float NdotV = max(0.0001f, dot(N, V));

    // ”½Ë•ûŒü
    float3 R = reflect(-V, N);

    // roughness‚É‰‚¶‚ÄMipƒŒƒxƒ‹‚ğ‘I‘ğ
    uint width, height, mipCount;
    prefilteredMap.GetDimensions(0, width, height, mipCount);
    float mipLevel = roughness * (float) (mipCount - 1);

    // –‘OƒtƒBƒ‹ƒ^ƒŠƒ“ƒOÏ‚İŠÂ‹«ƒ}ƒbƒv‚ğƒTƒ“ƒvƒŠƒ“ƒO
    float3 prefilteredColor = prefilteredMap.SampleLevel(
        samplerState, R, mipLevel).rgb;

    // BRDF LUT‚ğƒTƒ“ƒvƒŠƒ“ƒO
    float2 brdf = brdfLut.Sample(samplerState,
        float2(NdotV, roughness)).rg;

    // ƒtƒŒƒlƒ‹€
    float3 F = F0 + (1.0f - F0) * pow(clamp(1.0f - NdotV, 0.0f, 1.0f), 5.0f);
    
    return prefilteredColor * (F * brdf.x + brdf.y);
}

// Œõ‚ğ“–‚Ä‚½F‚ğ•Ô‚·
float3 ShadeSurface(
    SurfaceData s,
    TextureCube diffuseIem,
    TextureCube specularPmrem,
    Texture2D lutGgx,
    SamplerState samplerState)
{
    // PBR ƒpƒ‰ƒ[ƒ^[
    float3 diffuse_reflectance = lerp(s.albedo, 0.0f, s.metalness);
    float3 F0 = lerp(0.04f, s.albedo, s.metalness);

    float3 N = s.normal;
    float3 V = normalize(cameraPosition.xyz - s.worldPos);
    float3 diffuse = 0;
    float3 specular = 0;

    // •½sŒõŒ¹‚Ì‰e
    float shadow = CalcShadow(s.worldPos, N);
    
// •½sŒõŒ¹
    {
        float3 L = normalize(-directionalLight.direction.xyz);
        float3 LC = directionalLight.color.rgb * directionalLight.color.a * shadow;
        float3 d = 0;
        float3 sp = 0;
        DirectBRDF(diffuse_reflectance, F0, N, V, L, LC, s.roughness, d, sp);
        diffuse += d;
        specular += sp;
    }

    // ƒ|ƒCƒ“ƒgƒ‰ƒCƒg
    for (int i = 0; i < pointLightCount; i++)
    {
        float3 toLight = pointLights[i].position.xyz - s.worldPos;
        float dist = length(toLight);
        float range = pointLights[i].position.w;

        if (dist > range)
            continue;

        float attenuation = 1.0f - saturate(dist / range);
        attenuation *= attenuation;

        float3 L = normalize(toLight);
        float3 LC = pointLights[i].color.rgb
                  * pointLights[i].color.a
                  * attenuation;

        float3 d = 0;
        float3 sp = 0;
        DirectBRDF(diffuse_reflectance, F0, N, V, L, LC, s.roughness, d, sp);
        diffuse += d;
        specular += sp;
    }

    // ŠÔÚŒõiIBLj
    float3 iblDiffuse = IBL_Diffuse(
        diffuseIem, samplerState, N,
        diffuse_reflectance, F0, V);

    float3 iblSpecular = IBL_Specular(
        specularPmrem, lutGgx, samplerState,
        N, V, F0, s.roughness);

    float3 ambient = (iblDiffuse + iblSpecular) * iblIntensity;
    
    // Occlusion
    ambient *= s.occlusion;
    
    // ‰e‚Ì’†‚ÍŠÂ‹«Œõ‚àˆÃ‚­‚·‚é
    ambient *= lerp(1.0f, shadow, shadowStrength);
    
    return diffuse + specular + ambient + s.emissive;
}

#ifndef PBR_FUNCTIONS_HLSLI
#define PBR_FUNCTIONS_HLSLI

#ifndef PI
#define PI 3.14159265358979323846f
#endif

#ifndef GAMMA
#define GAMMA 2.2f

// ƒtƒŒƒlƒ‹€ (Schlick‚Ì‹ß—®)

// F0    : ‚’¼“üË‚Ì”½Ë—¦
// VdotH : ‹üƒxƒNƒgƒ‹‚Æƒn[ƒtƒxƒNƒgƒ‹‚Ì“àÏ
float3 CalcFresnel(float3 F0, float VdotH)
{
    return F0 + (1.0f - F0) * pow(clamp(1.0f - VdotH, 0.0f, 1.0f), 5.0f);
}

// ŠgU”½ËBRDF (³‹K‰»ƒ‰ƒ“ƒo[ƒg)

// VdotH              : ‹ü‚Æƒn[ƒtƒxƒNƒgƒ‹‚Ì“àÏ
// F0                 : ‚’¼“üË‚ÌƒtƒŒƒlƒ‹”½Ë—¦
// diffuse_reflectance: ŠgU”½Ë—¦
float3 DiffuseBRDF(float VdotH, float3 F0, float3 diffuse_reflectance)
{
    return (1.0f - CalcFresnel(F0, VdotH)) * (diffuse_reflectance / PI);
}

// D€: –@ü•ª•zŠÖ” (GGX)

// NdotH    : –@ü‚Æƒn[ƒtƒxƒNƒgƒ‹‚Ì“àÏ
// roughness: ‘e‚³
float CalcNDF(float NdotH, float roughness)
{
    float a = roughness * roughness;
    float a2 = a * a;
    float d = (NdotH * NdotH) * (a2 - 1.0f) + 1.0f;
    return a2 / (PI * d * d);
}

// G€: Šô‰½Œ¸Š€ (Smith-Schlick)

// NdotL    : –@ü‚ÆŒõŒ¹ƒxƒNƒgƒ‹‚Ì“àÏ
// NdotV    : –@ü‚Æ‹üƒxƒNƒgƒ‹‚Ì“àÏ
// roughness: ‘e‚³
float CalcGeometry(float NdotL, float NdotV, float roughness)
{
    float r = roughness * 0.5f;
    float shadowing = NdotL / (NdotL * (1.0f - r) + r);
    float masking = NdotV / (NdotV * (1.0f - r) + r);
    return shadowing * masking;
}

// ‹¾–Ê”½ËBRDF (Cook-Torranceƒ‚ƒfƒ‹)

// NdotV    : –@ü‚Æ‹üƒxƒNƒgƒ‹‚Ì“àÏ
// NdotL    : –@ü‚ÆŒõŒ¹ƒxƒNƒgƒ‹‚Ì“àÏ
// NdotH    : –@ü‚Æƒn[ƒtƒxƒNƒgƒ‹‚Ì“àÏ
// VdotH    : ‹ü‚Æƒn[ƒtƒxƒNƒgƒ‹‚Ì“àÏ
// F0       : ‚’¼“üË‚ÌƒtƒŒƒlƒ‹”½Ë—¦
// roughness: ‘e‚³
float3 SpecularBRDF(float NdotV, float NdotL, float NdotH, float VdotH,
                    float3 F0, float roughness)
{
    float D = CalcNDF(NdotH, roughness);
    float G = CalcGeometry(NdotL, NdotV, roughness);
    float3 F = CalcFresnel(F0, VdotH);
    return (D * G * F) / (NdotL * NdotV * 4.0f);
}

// ’¼ÚŒõ‚ÌPBRƒ‰ƒCƒeƒBƒ“ƒO

// diffuse_reflectance: ŠgU”½Ë—¦
// F0                 : ‚’¼“üË‚ÌƒtƒŒƒlƒ‹”½Ë—¦
// N                  : –@ü (³‹K‰»Ï‚İ)
// V                  : ‹üƒxƒNƒgƒ‹ (ƒJƒƒ‰•ûŒü, ³‹K‰»Ï‚İ)
// L                  : ŒõŒ¹•ûŒüƒxƒNƒgƒ‹ (ŒõŒ¹‚ÖŒü‚©‚¤, ³‹K‰»Ï‚İ)
// light_color        : ƒ‰ƒCƒgƒJƒ‰[
// roughness          : ‘e‚³
void DirectBRDF(
    float3 diffuse_reflectance,
    float3 F0,
    float3 N, float3 V, float3 L,
    float3 light_color,
    float roughness,
    out float3 out_diffuse,
    out float3 out_specular)
{
    float3 H = normalize(L + V);
    float NdotV = max(0.0001f, dot(N, V));
    float NdotL = max(0.0001f, dot(N, L));
    float NdotH = max(0.0001f, dot(N, H));
    float VdotH = max(0.0001f, dot(V, H));

    float3 irradiance = light_color * NdotL;

    out_diffuse = DiffuseBRDF(VdotH, F0, diffuse_reflectance) * irradiance;
    out_specular = SpecularBRDF(NdotV, NdotL, NdotH, VdotH, F0, roughness) * irradiance;
}

    // ‰eƒ}ƒbƒv1ƒsƒNƒZƒ‹‚ªƒ[ƒ‹ƒh‚Å‰½m‚©
    uint width, height, elements;
    shadowMap.GetDimensions(width, height, elements);
    float texelWorld = 2.0f / (length(float3(m._11, m._21, m._31)) * width);
    
    // •\–Ê‚ğ–@ü‚Ì•ûŒü‚Ö­‚µ•‚‚©‚¹‚Ä‚©‚ç’²‚×‚é
    float3 offsetPos = worldPos + normal * texelWorld * SHADOW_NORMAL_OFFSET;
    
    // ƒ‰ƒCƒg‚©‚çŒ©‚½ˆÊ’u‚Ö
    float4 lightPos = mul(float4(offsetPos, 1.0f), m);
    float3 ndc = lightPos.xyz / lightPos.w;
    
    // NDC ‚©‚ç ƒeƒNƒXƒ`ƒƒÀ•W ‚Ö•ÏŠ·
    float2 uv = float2(ndc.x * 0.5f + 0.5f, -ndc.y * 0.5f + 0.5f);
    float depth = ndc.z;
    
    // ƒ‰ƒCƒg‚Ì”ÍˆÍ‚æ‚è‰œ‚ÍA‰eƒ}ƒbƒv‚Éî•ñ‚ª‚È‚¢‚Ì‚Å‰e‚È‚µ‚É‚·‚é
    if (depth > 1.0f)
    {
        return 1.0f;
    }
    
    // ­‚µè‘O‚ÉŠñ‚¹‚½A©•ª‚Ì[“x
    float compareDepth = depth - SHADOW_DEPTH_BIAS;

    // ‰eƒ}ƒbƒv1ƒsƒNƒZƒ‹‚Ô‚ñ‚Ì UV ‚Ì‘å‚«‚³
    float2 texelUV = 1.0f / float2(width, height);

    // ü‚è‚ÌƒsƒNƒZƒ‹‚à“Ç‚ñ‚Å•½‹Ï‚·‚é
    float sum = 0.0f;
    [unroll]
    for (int y = -SHADOW_PCF_RADIUS; y <= SHADOW_PCF_RADIUS; y++)
    {
        [unroll]
        for (int x = -SHADOW_PCF_RADIUS; x <= SHADOW_PCF_RADIUS; x++)
        {
            float2 offsetUV = uv + float2(x, y) * texelUV;
            sum += shadowMap.SampleCmpLevelZero(shadowSampler, float3(offsetUV, (float) cascade), compareDepth);
        }
    }

    const float sampleCount = (SHADOW_PCF_RADIUS * 2 + 1) * (SHADOW_PCF_RADIUS * 2 + 1);
    return sum / sampleCount;
}

// Œõ‚ª“Í‚¢‚Ä‚¢‚ê‚Î 1A‰e‚È‚ç 0 ‚ğ•Ô‚·
float CalcShadow(float3 worldPos, float3 normal)
{
    // ƒJƒƒ‰‚©‚ç‚Ì‹——£
    float viewDepth = mul(float4(worldPos, 1.0f), view).z;
    
    // ‰e‚ğo‚·”ÍˆÍ‚æ‚è‰œ‚ÍA‰e‚È‚µ
    float shadowEnd = cascadeSplits[SHADOW_CASCADE_COUNT - 1];
    if (viewDepth >= shadowEnd)
    {
        return 1.0f;
    }
    
    // ‚Ç‚Ì’i‚ğ“Ç‚Ş‚©
    int cascade = 0;
    [unroll]
    for (int i = 0; i < SHADOW_CASCADE_COUNT - 1; i++)
    {
        if (viewDepth > cascadeSplits[i])
        {
            cascade = i + 1;
        }
    }
    
    float shadow = SampleShadowCascade(cascade, worldPos, normal);

    // ’i‚Ì‹«–Ú
    if (cascade < SHADOW_CASCADE_COUNT - 1)
    {
        float cascadeStart = (cascade == 0) ? 0.0f : cascadeSplits[cascade - 1];
        float cascadeEnd = cascadeSplits[cascade];
        float blendStart = cascadeEnd - (cascadeEnd - cascadeStart) * CASCADE_BLEND_RATIO;
        float blend = saturate((viewDepth - blendStart) / (cascadeEnd - blendStart));

        // ¬‚º‚é‹æŠÔ‚É“ü‚Á‚½Š‚¾‚¯AŸ‚Ì’i‚à“Ç‚Ş
        if (blend > 0.0f)
        {
            float nextShadow = SampleShadowCascade(cascade + 1, worldPos, normal);
            shadow = lerp(shadow, nextShadow, blend);
        }
    }
    
    // ‚¢‚¿‚Î‚ñ‰œ‚Ìè‘O‚ÅA‰e‚ğ­‚µ‚¸‚ÂÁ‚·
    float fadeStart = shadowEnd * (1.0f - SHADOW_FADE_RATIO);
    float fade = saturate((viewDepth - fadeStart) / (shadowEnd - fadeStart));
    return lerp(shadow, 1.0f, fade);
}
